0.1 Purpose

This document constructs the supplementary between-subject benchmark using only session 1 outcomes. That benchmark emulates a standard parallel-arm experiment by comparing participants assigned to high_tempo first versus low_tempo first before any second-session carryover can occur.

0.2 Load benchmark sample

flow <- sanitize_dt_ascii(fread(derived_path("participant_flow.csv")))
long_dt <- sanitize_dt_ascii(fread(derived_path("long_repeated_measures.csv")))

bench <- long_dt[
  in_first_period_benchmark == 1L & period == 1L
]

bench[, treatment_label := fifelse(high_tempo == 1L, "High tempo first", "Low tempo first")]

stopifnot(nrow(bench) == 46L)
stopifnot(uniqueN(bench$participant_id) == nrow(bench))

0.4 Balance checks

bench[, active_at_least_weekly := as.integer(activity_level_baseline %in% c(3, 4))]

  ,
  .(
    n = .N,
    mean_age = mean(participant_age, na.rm = TRUE),

It is less threatening for the primary estimate because the main analysis uses the crossover design and participant fixed effects, which absorb stable participant-level differences.

0.7 Benchmark Model 1: raw contrast

t_bench <- t.test(outcome_tte ~ high_tempo, data = bench)

mean_high <- bench[high_tempo == 1L, mean(outcome_tte)]
mean_low <- bench[high_tempo == 0L, mean(outcome_tte)]
diff_raw <- mean_high - mean_low

raw_results <- data.table(
  quantity = c(
    "Mean high-first session 1 TTE",
    "Mean low-first session 1 TTE",
    "Difference (high first - low first)",
    "95% CI lower",
    "95% CI upper",
    "t statistic",
    "p-value"
  ),
  value = c(
    mean_high,
    mean_low,
    diff_raw,
    t_bench$conf.int[1],
    t_bench$conf.int[2],
    unname(t_bench$statistic),
    t_bench$p.value
  )
)
raw_results[, value := round(value, 4)]

0.8 Benchmark Model 2: precision-adjusted regression

bench[, participant_gender := factor(participant_gender)]

adj_fit <- lm(
  outcome_tte ~ high_tempo + participant_age + participant_gender,
  data = bench
)

adj_sum <- summary(adj_fit)$coefficients

adj_results <- data.table(
  term = c("High-tempo first", "Participant age"),
  estimate = c(adj_sum["high_tempo", "Estimate"], adj_sum["participant_age", "Estimate"]),
  std_error = c(adj_sum["high_tempo", "Std. Error"], adj_sum["participant_age", "Std. Error"]),
  p_value = c(adj_sum["high_tempo", "Pr(>|t|)"], adj_sum["participant_age", "Pr(>|t|)"])
)

adj_results[, `:=`(
  estimate = round(estimate, 4),
  std_error = round(std_error, 4),
  p_value = round(p_value, 4)
)]

kable(adj_results, caption = "Benchmark Model 2. Precision-adjusted regression")
