0.1 Purpose

This document checks whether broad period effects or order patterns appear large enough to threaten interpretation of the main crossover estimate.

These diagnostics are descriptive first. They do not replace the primary model; they help explain whether session order or repeated exposure appears material and whether the period term in the fixed-effects model is doing useful work.

period_check <- t.test(

0.6.2 Sequence diagnostic regression

long_primary[, crossover_order := factor(crossover_order)]
long_primary[, period_factor := factor(period)]

diag_fit <- lm(
  outcome_tte ~ period_factor * crossover_order,
  data = long_primary
)

diag_vcov <- sandwich::vcovCL(diag_fit, cluster = ~ participant_id, type = "HC1")
diag_ct <- lmtest::coeftest(diag_fit, vcov. = diag_vcov)

diag_terms <- intersect(
  c("period_factor2", "crossover_orderlow_tempo_first", "period_factor2:crossover_orderlow_tempo_first"),
  rownames(diag_ct)
)

diag_results <- data.table(
  term = diag_terms,
  estimate = diag_ct[diag_terms, "Estimate"],
  std_error = diag_ct[diag_terms, "Std. Error"],
  p_value = diag_ct[diag_terms, "Pr(>|t|)"]
)
diag_results[, `:=`(
  estimate = round(estimate, 4),
  std_error = round(std_error, 4),
  p_value = round(p_value, 4)
)]

kable(diag_results, caption = "Sequence diagnostic regression")
